-- 3. Нормализация / стоп-лист. Возвращает только boolean; таблицу блокировок не раскрывает.
create or replace function public.scbb_normalize_nick(p_nick text)
returns text
language sql immutable parallel safe
as $$
  select regexp_replace(replace(lower(coalesce(p_nick, '')), 'ё', 'е'), '[^[:alnum:]]', '', 'g');
$$;

create or replace function public.is_nick_allowed(p_nick text)
returns boolean
language plpgsql stable security definer
set search_path = public
as $$
declare
  v_normalized text;
  v_words jsonb := '[]'::jsonb;
  v_word text;
  v_key text;
begin
  v_normalized := public.scbb_normalize_nick(p_nick);
  if length(v_normalized) < 2 or length(v_normalized) > 24 then return false; end if;
  if exists (select 1 from public.nick_blocks b where b.normalized_nick = v_normalized) then return false; end if;
  select c.value::jsonb into v_words from public.config c where c.key = 'nickname_stoplist';
  v_words := coalesce(v_words, '[]'::jsonb);
  if jsonb_typeof(v_words) <> 'array' then v_words := '[]'::jsonb; end if;
  for v_word in select jsonb_array_elements_text(v_words) loop
    v_key := public.scbb_normalize_nick(v_word);
    if length(v_key) > 0 and position(v_key in v_normalized) > 0 then return false; end if;
  end loop;
  return true;
exception when others then
  -- При ошибке стоп-листа новые рекорды не принимаются (fail closed).
  return false;
end;
$$;

create or replace function public.scbb_sync_record_nick()
returns trigger
language plpgsql
security definer
set search_path = public
as $$
begin
  if current_setting('app.scbb_migration_mode', true) = 'on' then
    new.nick := coalesce(nullif(trim(new.nick), ''), nullif(trim(new.player_name), ''), 'Игрок');
    new.player_name := coalesce(nullif(trim(new.player_name), ''), new.nick, 'Игрок');
    return new;
  end if;
  if tg_op = 'UPDATE' then
    if new.nick is distinct from old.nick and new.player_name is not distinct from old.player_name then
      new.player_name := new.nick;
    elsif new.player_name is distinct from old.player_name and new.nick is not distinct from old.nick then
      new.nick := new.player_name;
    else
      new.nick := coalesce(nullif(trim(new.nick), ''), nullif(trim(new.player_name), ''), 'Игрок');
      new.player_name := coalesce(nullif(trim(new.player_name), ''), new.nick, 'Игрок');
    end if;
  else
    -- Старый автомат отправляет только player_name. Поле nick имеет default='Игрок',
    -- поэтому для legacy INSERT значение player_name должно иметь приоритет.
    if nullif(trim(new.player_name), '') is not null and (new.nick is null or trim(new.nick) = '' or trim(new.nick) = 'Игрок') then
      new.nick := new.player_name;
    end if;
    new.nick := coalesce(nullif(trim(new.nick), ''), nullif(trim(new.player_name), ''), 'Игрок');
    new.player_name := coalesce(nullif(trim(new.player_name), ''), new.nick, 'Игрок');
  end if;
  if not public.is_nick_allowed(new.nick) then
    raise exception 'NICK_NOT_ALLOWED' using errcode = '23514';
  end if;
  return new;
end;
$$;

drop trigger if exists trg_scbb_record_nick on public.records;
create trigger trg_scbb_record_nick
before insert or update of nick, player_name on public.records
for each row execute function public.scbb_sync_record_nick();

-- Серверная модерация одним вызовом. Разрешена только service_role (Edge Function).
create or replace function public.moderate_records_by_nick(p_record_id bigint, p_mode text, p_admin_id uuid default null)
returns integer
language plpgsql security definer
set search_path = public
as $$
declare
  v_nick text;
  v_norm text;
  v_count integer;
begin
  if p_mode not in ('hidden', 'banned') then raise exception 'INVALID_MODE'; end if;
  select coalesce(nick, player_name) into v_nick from public.records where id = p_record_id;
  if v_nick is null then raise exception 'RECORD_NOT_FOUND'; end if;
  v_norm := public.scbb_normalize_nick(v_nick);
  insert into public.nick_blocks(normalized_nick, nick, block_type, reason, created_by)
  values (v_norm, v_nick, p_mode, 'Модерация панели СибКодБокс', p_admin_id)
  on conflict (normalized_nick) do update set nick = excluded.nick, block_type = excluded.block_type, reason = excluded.reason, created_by = excluded.created_by;
  if p_mode = 'banned' then
    update public.records set banned = true, hidden = true where public.scbb_normalize_nick(coalesce(nick, player_name)) = v_norm;
  else
    update public.records set hidden = true where public.scbb_normalize_nick(coalesce(nick, player_name)) = v_norm;
  end if;
  get diagnostics v_count = row_count;
  return v_count;
end;
$$;

-- Атомарное одноразовое использование QR. Активный срок сессии начинается именно при сканировании.
create or replace function public.redeem_scbb_session(p_qr_token_hash text, p_session_key_hash text)
returns table(session_id uuid, owner_name text, expires_at timestamptz)
language plpgsql security definer
set search_path = public
as $$
declare v_row public.sessions%rowtype;
begin
  update public.sessions s
  set status = 'active', started_at = now(),
      expires_at = now() + make_interval(mins => s.minutes),
      session_key_hash = p_session_key_hash, updated_at = now()
  where s.qr_token_hash = p_qr_token_hash and s.status = 'issued' and s.qr_expires_at > now()
  returning s.* into v_row;
  if not found then return; end if;
  return query select v_row.id, v_row.owner_name, v_row.expires_at;
end;
$$;

-- Старый автомат имеет право поставить used=true, но не может повторно открыть used-билет.
create or replace function public.scbb_protect_legacy_ticket_used()
returns trigger language plpgsql security definer set search_path = public as $$
begin
  if auth.role() = 'anon' and old.used = true and new.used = false then
    raise exception 'TICKET_ALREADY_USED' using errcode = '23514';
  end if;
  return new;
end;
$$;
drop trigger if exists trg_scbb_legacy_ticket_used on public.tickets;
create trigger trg_scbb_legacy_ticket_used before update of used on public.tickets
for each row execute function public.scbb_protect_legacy_ticket_used();
revoke all on function public.scbb_protect_legacy_ticket_used() from public, anon, authenticated;

-- Transitional mirror: keeps the legacy config.games_json shape for old kiosks.
-- Unknown legacy properties (accent, bgUrl, badge, etc.) remain in games.legacy_json.
create or replace function public.scbb_refresh_games_json()
returns void
language plpgsql
security definer
set search_path = public
as $$
declare v_catalog jsonb;
begin
  select coalesce(jsonb_agg(
    coalesce(g.legacy_json, '{}'::jsonb) || jsonb_build_object(
      'id', g.id,
      'slug', g.slug,
      'title', g.title,
      'description', g.description,
      'desc', g.description,
      'genre', g.genre,
      'developer', g.developer,
      'author', g.developer,
      'iconUrl', g.icon_url,
      'icon_url', g.icon_url,
      'coverUrl', coalesce(g.legacy_json->>'coverUrl', g.icon_url),
      'url', g.entry_url,
      'entry_url', g.entry_url,
      'repoUrl', g.repo_url,
      'repo_url', g.repo_url,
      'badge', coalesce(g.legacy_json->>'badge', 'NEW'),
      'platform', coalesce(g.legacy_json->>'platform', 'github-folder'),
      'storage', 'github',
      'created_at', g.created_at
    ) order by g.created_at
  ), '[]'::jsonb)
  into v_catalog
  from public.games g
  where g.is_active = true;

  insert into public.config(key, value)
  values ('games_json', v_catalog::text)
  on conflict (key) do update set value = excluded.value;
end;
$$;

create or replace function public.scbb_mirror_games_json()
returns trigger
language plpgsql
security definer
set search_path = public
as $$
begin
  perform public.scbb_refresh_games_json();
  return null;
end;
$$;

drop trigger if exists trg_scbb_mirror_games_json on public.games;
create trigger trg_scbb_mirror_games_json
 after insert or update or delete on public.games
 for each statement execute function public.scbb_mirror_games_json();
revoke all on function public.scbb_refresh_games_json() from public, anon, authenticated;
